  <button onclick="initGame()">Рестарт</button>

<script>
  let cells, current, gameOver;
  const boardEl = document.getElementById('board');
  const statusEl = document.getElementById('status');

  function initGame() {
    cells = Array(9).fill('');
    current = 'X';
    gameOver = false;
    boardEl.innerHTML = '';
    statusEl.textContent = 'Ход крестиков';

    for (let i = 0; i < 9; i++) {
      const c = document.createElement('div');
      c.className = 'cell';
      c.addEventListener('click', () => move(i));
      boardEl.appendChild(c);
    }
  }

  function move(i) {
    if (gameOver || cells[i]) return;
    cells[i] = current;
    const cell = boardEl.children[i];
    cell.textContent = current;
    cell.classList.add('taken', current.toLowerCase());

    const win = [[0,1,2],[3,4,5],[6,7,8],[0,3,6],[1,4,7],[2,5,8],[0,4,8],[2,4,6]]
      .find(l => l.every(x => cells[x] === current));
    if (win) {
      gameOver = true;
      statusEl.textContent = current + ' победил!';
      return;
    }
    if (cells.every(x => x)) {
      gameOver = true;
      statusEl.textContent = 'Ничья';
      return;
    }
    current = current === 'X' ? 'O' : 'X';
    statusEl.textContent = 'Ход ' + (current === 'X' ? 'крестиков' : 'ноликов');
  }

  initGame();
</script>
